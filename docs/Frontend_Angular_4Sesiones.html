AuthService — login, token y signal de usuario

signal() almacena el usuario actual reactivamente: cualquier componente que lo lea se actualiza automáticamente al hacer login o logout. El token se guarda en localStorage.

src/app/features/auth/auth.service.ts 
import { Injectable, signal, computed, inject } from '@angular/core';
import { HttpClient, HttpHeaders } from '@angular/common/http';
import { Router } from '@angular/router';
import { tap } from 'rxjs/operators';
import { environment } from '../../../environments/environment';
import { LoginRequest, TokenResponse, Usuario } from '../../shared/models';

@Injectable({ providedIn: 'root' })
export class AuthService {
  private http   = inject(HttpClient);
  private router = inject(Router);

  // Signal reactivo: toda la app reacciona al cambio de usuario
  currentUser  = signal<Usuario | null>(null);
  isLoggedIn   = computed(() => this.currentUser() !== null);
  isAdmin      = computed(() => this.currentUser()?.rol === 'admin');

  login(credentials: LoginRequest) {
    // FastAPI OAuth2 espera form-data, no JSON
    const body = new URLSearchParams();
    body.set('username', credentials.username);
    body.set('password', credentials.password);

    return this.http.post<TokenResponse>(
      `${environment.apiUrl}/login`, body.toString(),
      { headers: new HttpHeaders({'Content-Type':'application/x-www-form-urlencoded'}) }
    ).pipe(
      tap(res => {
        localStorage.setItem('token', res.access_token);
        this.loadCurrentUser();
      })
    );
  }

  loadCurrentUser() {
    // Decodifica el payload del JWT sin librería externa
    const token = localStorage.getItem('token');
    if (!token) return;
    try {
      const payload = JSON.parse(atob(token.split('.')[1]));
      // Reconstruir objeto usuario mínimo desde el payload
      this.currentUser.set({ id: 0, nombre: '', email: payload.sub,
                             rol: payload.rol });
    } catch { this.logout(); }
  }

  logout() {
    localStorage.removeItem('token');
    this.currentUser.set(null);
    this.router.navigate(['/login']);
  }

  getToken(): string | null {
    return localStorage.getItem('token');
  }
}

JWT Interceptor — agrega Bearer a cada petición

El interceptor funciona como un middleware Angular: intercepta TODAS las peticiones HTTP salientes y agrega el header Authorization automáticamente. Los componentes no tienen que preocuparse por el token.

src/app/core/interceptors/auth.interceptor.ts 
import { HttpInterceptorFn } from '@angular/common/http';
import { inject } from '@angular/core';
import { AuthService } from '../../features/auth/auth.service';

export const authInterceptor: HttpInterceptorFn = (req, next) => {
  const auth  = inject(AuthService);
  const token = auth.getToken();

  if (token) {
    // Clonar la petición con el header Authorization
    const authReq = req.clone({
      setHeaders: { Authorization: `Bearer ${token}` }
    });
    return next(authReq);
  }
  return next(req);
};

AuthGuard — proteger rutas

CanActivateFn es la API moderna funcional de Angular 17+. Si el usuario no tiene token, redirige al login antes de renderizar el componente.

import { CanActivateFn, Router } from '@angular/router';
import { inject } from '@angular/core';
import { AuthService } from '../../features/auth/auth.service';

export const authGuard: CanActivateFn = () => {
  const auth   = inject(AuthService);
  const router = inject(Router);

  if (auth.isLoggedIn()) return true;
  return router.createUrlTree(['/login']);
};

// Guard adicional para rutas solo de admin

  const auth   = inject(AuthService);

  return router.createUrlTree(['/dashboard']);

Autoevaluación · Estación 02

1. ¿Por qué el interceptor "clona" la petición en lugar de modificarla directamente?

💡 HttpRequest es inmutable en Angular. clone() crea una copia con los headers adicionales sin tocar el original.

💡 Los signals son reactivos: el cambio se propaga automáticamente a todos los consumidores, incluyendo la vista.

3. ¿Por qué el login envía form-data y no JSON?

💡 El endpoint /login de FastAPI usa OAuth2PasswordRequestForm que lee los campos del body como form-data con Content-Type: application/x-www-form-urlencoded.

HttpErrorInterceptor — manejo global de errores

Centraliza el manejo de errores HTTP: 401 redirige al login automáticamente, 403 muestra aviso de permisos, 5xx muestra error de servidor. Los componentes no necesitan manejar cada caso.

src/app/core/interceptors/error.interceptor.ts 
import { HttpInterceptorFn } from '@angular/common/http';
import { inject } from '@angular/core';
import { catchError, throwError } from 'rxjs';
import { Router } from '@angular/router';
import { AuthService } from '../../features/auth/auth.service';

export const errorInterceptor: HttpInterceptorFn = (req, next) => {
  const router = inject(Router);
  const auth   = inject(AuthService);

  return next(req).pipe(
    catchError(err => {
      switch (err.status) {
        case 401:
          auth.logout();   // token expirado → redirige al login
          break;
        case 403:
          console.warn('Acceso denegado:', err.error?.detail);
          router.navigate(['/dashboard']);
          break;
        case 0:
          console.error('Sin conexión con el backend');
          break;
      }
      return throwError(() => err);
    })
  );
};

DepartamentosService — patrón CRUD completo

Observable para cada operación CRUD. El servicio no suscribe: devuelve el Observable para que el componente lo gestione. signal() almacena la lista localmente para evitar re-fetches innecesarios.

src/app/shared/services/departamentos.service.ts 
import { Injectable, signal, inject } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { tap } from 'rxjs/operators';
import { environment } from '../../../environments/environment';
import { Departamento } from '../models';

@Injectable({ providedIn: 'root' })
export class DepartamentosService {
  private http    = inject(HttpClient);
  private baseUrl = `${environment.apiUrl}/departamentos`;

  // Signal local: caché reactivo de la lista
  departamentos = signal<Departamento[]>([]);

  getAll() {
    return this.http.get<Departamento[]>(this.baseUrl).pipe(
      tap(lista => this.departamentos.set(lista))
    );
  }

  getById(id: number) {
    return this.http.get<Departamento>(`${this.baseUrl}/${id}`);
  }

  create(data: { nombre: string; codigo: string }) {
    return this.http.post<Departamento>(this.baseUrl, data).pipe(
      tap(nuevo => this.departamentos.update(l => [...l, nuevo]))
    );
  }

  update(id: number, data: Partial<Departamento>) {
    return this.http.put<Departamento>(`${this.baseUrl}/${id}`, data).pipe(
      tap(actualizado =>
        this.departamentos.update(l =>
          l.map(d => d.id === id ? actualizado : d)
        )
      )
    );
  }

  delete(id: number) {
    return this.http.delete(`${this.baseUrl}/${id}`).pipe(
      tap(() => this.departamentos.update(l => l.filter(d => d.id !== id)))
    );
  }
}

DepartamentosComponent — lista + formulario inline

src/app/features/departamentos/departamentos.component.ts 
import { Component, OnInit, signal, inject } from '@angular/core';
import { CommonModule } from '@angular/common';
import { ReactiveFormsModule, FormBuilder, Validators } from '@angular/forms';
import { DepartamentosService } from '../../shared/services/departamentos.service';
import { AuthService } from '../auth/auth.service';
import { Departamento } from '../../shared/models';

@Component({
  selector: 'app-departamentos',
  standalone: true,
  imports: [CommonModule, ReactiveFormsModule],
  templateUrl: './departamentos.component.html'
})
export class DepartamentosComponent implements OnInit {
  private svc  = inject(DepartamentosService);
  auth         = inject(AuthService);      // para isAdmin en la plantilla

  // Re-exportar el signal del servicio a la plantilla
  departamentos = this.svc.departamentos;

  editando  = signal<Departamento | null>(null);
  loading   = signal(false);
  error     = signal('');

  private fb = inject(FormBuilder);
  form = this.fb.group({
    nombre: ['', Validators.required],
    codigo: ['', [Validators.required, Validators.maxLength(10)]]
  });

  ngOnInit() {
    this.svc.getAll().subscribe();
  }

  guardar() {
    if (this.form.invalid) return;
    this.loading.set(true);
    const data = this.form.value as { nombre: string; codigo: string };
    const op = this.editando()
      ? this.svc.update(this.editando()!.id, data)
      : this.svc.create(data);

    op.subscribe({
      next: () => { this.form.reset(); this.editando.set(null); this.loading.set(false); },
      error: (e) => { this.error.set(e.error?.detail ?? 'Error'); this.loading.set(false); }
    });
  }

  editar(d: Departamento) {
    this.editando.set(d);
    this.form.patchValue({ nombre: d.nombre, codigo: d.codigo });
  }

  cancelar() {
    this.editando.set(null);
    this.form.reset();
  }

  eliminar(id: number) {
    if (!confirm('¿Eliminar este departamento?')) return;
    this.svc.delete(id).subscribe({
      error: (e) => this.error.set(e.error?.detail ?? 'Error al eliminar')
    });
  }
}

Autoevaluación · Estación 03

1. ¿Por qué el servicio NO se suscribe internamente y devuelve el Observable?

💡 El componente sabe cuándo destruirse y cancela la suscripción. El servicio no sabe cuánto tiempo vivirá quien lo usa.

2. ¿Qué hace tap() en el pipe del Observable?

💡 tap() es un operador "de escucha": deja pasar el valor sin tocarlo y aprovecha para actualizar el caché local (el signal).

3. ¿Qué diferencia hay entre patchValue() y setValue() en un formulario reactivo?

💡 patchValue() es más flexible para edición parcial. setValue() lanza error si falta algún campo del grupo.

import { Component, OnInit, signal, inject, computed } from '@angular/core';
import { CommonModule } from '@angular/common';
import { ReactiveFormsModule, FormBuilder, Validators,
         AbstractControl, ValidationErrors } from '@angular/forms';
import { CursosService }   from '../../shared/services/cursos.service';
import { UsuariosService } from '../../shared/services/usuarios.service';
import { AuthService }     from '../auth/auth.service';

// Validator personalizado: fecha_inscripcion <= fecha_inicio
function fechaInscripcionValidator(control: AbstractControl): ValidationErrors | null {
  const form   = control.parent;
  if (!form) return null;
  const fechaInsc  = form.get('fecha_inscripcion')?.value;
  const fechaInicio = form.get('fecha_inicio')?.value;
  if (fechaInsc && fechaInicio && fechaInsc > fechaInicio) {
    return { fechaPosterior: true };
  }
  return null;
}

  selector: 'app-cursos',

export class CursosComponent implements OnInit {

  private usuSvc   = inject(UsuariosService);

  cursoSeleccionado = signal<Curso | null>(null);
  loading  = signal(false);
  error    = signal('');
  exito    = signal('');

  formCurso = this.fb.group({
    nombre:      ['', Validators.required],
    descripcion: [''],
    capacidad:   [30, [Validators.required, Validators.min(1)]],
    fecha_inicio: ['', Validators.required]
  });

  formInscripcion = this.fb.group({
    usuario_id:       [null as number | null, Validators.required],
    fecha_inscripcion:['', [Validators.required, fechaInscripcionValidator]],
    fecha_inicio:     ['']  // campo oculto para el validator cruzado
  });

  ngOnInit() {
    this.cursoSvc.getAll().subscribe();
    this.usuSvc.getAll().subscribe();
  }

  crearCurso() {
    if (this.formCurso.invalid) return;
    this.loading.set(true);
    this.cursoSvc.create(this.formCurso.value as any).subscribe({
      next: () => { this.formCurso.reset({ capacidad: 30 });
                    this.loading.set(false); this.exito.set('Curso creado'); },
      error: (e) => { this.error.set(e.error?.detail ?? 'Error');
                      this.loading.set(false); }
    });
  }

  seleccionarCurso(c: Curso) {
    this.cursoSeleccionado.set(c);
    this.cursoSvc.getInscritos(c.id).subscribe();
    // Sincronizar la fecha del curso al formulario de inscripción
    this.formInscripcion.patchValue({ fecha_inicio: c.fecha_inicio });
  }

  inscribir() {
    if (this.formInscripcion.invalid) return;
    const { usuario_id, fecha_inscripcion } = this.formInscripcion.value;
    const cursoId = this.cursoSeleccionado()!.id;
    this.loading.set(true);

    this.cursoSvc.inscribir(cursoId, usuario_id!, fecha_inscripcion!).subscribe({
      next: () => {
        this.exito.set('Inscripción exitosa');
        this.cursoSvc.getInscritos(cursoId).subscribe();
        this.formInscripcion.reset({ fecha_inicio: fecha_inscripcion });
        this.loading.set(false);
      },
      error: (e) => { this.error.set(e.error?.detail ?? 'Error');
                      this.loading.set(false); }
    });
  }

  desinscribir(usuarioId: number) {
    const cursoId = this.cursoSeleccionado()!.id;
    this.cursoSvc.desinscribir(cursoId, usuarioId).subscribe({
      next: () => this.cursoSvc.getInscritos(cursoId).subscribe(),
      error: (e) => this.error.set(e.error?.detail ?? 'Error')
    });
  }

  // Cupo disponible como computed
  cupoDisponible = computed(() => {
    const c = this.cursoSeleccionado();
    return c ? c.capacidad - this.inscritos().length : 0;
  });
}

                         cursos: c.length });
        this.loading.set(false);
      },
      error: () => this.loading.set(false)
    });
  }

import { Component, inject } from '@angular/core';
import { CommonModule } from '@angular/common';
import { RouterModule, ActivatedRoute, Router,
         NavigationEnd } from '@angular/router';
import { filter, map } from 'rxjs/operators';
import { toSignal } from '@angular/core/rxjs-interop';

@Component({
  selector: 'app-breadcrumb',
  standalone: true,
  imports: [CommonModule, RouterModule],
  template: `
    <nav class="breadcrumb" aria-label="breadcrumb">
      <ol>
        <li><a routerLink="/dashboard">Inicio</a></li>
        <li *ngFor="let crumb of migas(); let last = last">
          <span class="sep">›</span>
          <a *ngIf="!last" [routerLink]="crumb.url">{{ crumb.label }}</a>
          <span *ngIf="last" class="activo">{{ crumb.label }}</span>
        </li>
      </ol>
    </nav>
  `,
  styles: [`
    .breadcrumb ol { display: flex; align-items: center;
                     gap: 6px; list-style: none;
                     font-size: 13px; color: var(--suave); }
    .sep { opacity: .5; }
    .activo { color: var(--tinta); font-weight: 600; }
    a { color: var(--azul); text-decoration: none; }
    a:hover { text-decoration: underline; }
  `]
})
export class BreadcrumbComponent {
  private router = inject(Router);

  // Construye las migas desde la URL activa
  migas = toSignal(
    this.router.events.pipe(
      filter(e => e instanceof NavigationEnd),
      map(() => {
        const partes = this.router.url.split('/').filter(Boolean);
        const labels: Record<string, string> = {
          dashboard: 'Dashboard', usuarios: 'Usuarios',
          departamentos: 'Departamentos', cursos: 'Cursos'
        };
        return partes.map((p, i) => ({
          label: labels[p] ?? p,
          url: '/' + partes.slice(0, i + 1).join('/')
        }));
      })
    ),
    { initialValue: [] }
  );
}

import { Injectable, signal, effect } from '@angular/core';

@Injectable({ providedIn: 'root' })
export class ThemeService {
  // Recuperar preferencia guardada (o preferencia del sistema)
  private inicial: 'dark' | 'light' =
    (localStorage.getItem('tema') as any) ??
    (window.matchMedia('(prefers-color-scheme: dark)').matches
      ? 'dark' : 'light');

  tema = signal<'dark' | 'light'>(this.inicial);

  constructor() {
    // effect() reacciona automáticamente cuando tema() cambia
    effect(() => {
      const t = this.tema();
      document.documentElement.classList.toggle('dark', t === 'dark');
      localStorage.setItem('tema', t);
    });
    // Aplicar al arranque
    document.documentElement.classList.toggle('dark', this.inicial === 'dark');
  }

  toggle() {
    this.tema.update(t => t === 'dark' ? 'light' : 'dark');
  }
}

1. ¿Por qué se necesita un 404.html en GitHub Pages para Angular?

💡 Las SPAs manejan el routing en el navegador (client-side). Cualquier URL directa que no sea / devuelve 404 en servidores estáticos. La solución es redirigir todas las rutas al index.html.

💡 Los signals están integrados en el scheduler de Angular. Los componentes que usan signals se actualizan quirúrgicamente solo cuando sus signals cambian, sin revisar el árbol completo.
