from sqlalchemy import select as sa_select

async def obtener_inscritos_con_fecha(db: AsyncSession, curso_id: int):
    # La tabla pivote tiene la fecha_inscripcion como columna propia
    rows = (await db.execute(
        sa_select(usuario_curso)
        .where(usuario_curso.c.curso_id == curso_id)
    )).fetchall()
    return [
        {
            "usuario_id":        r.usuario_id,
            "fecha_inscripcion": str(r.fecha_inscripcion)
        }
        for r in rows
    ]

async def desinscribir_usuario(
    db: AsyncSession, curso_id: int, usuario_id: int
):
    from sqlalchemy import delete as sa_delete
    result = await db.execute(
        sa_delete(usuario_curso)
        .where(usuario_curso.c.curso_id   == curso_id)
        .where(usuario_curso.c.usuario_id == usuario_id)
    )
    await db.commit()
    return result.rowcount > 0   # True si se eliminó, False si no existía

import os
from dotenv import load_dotenv

load_dotenv()

SECRET_KEY = os.getenv("JWT_SECRET_KEY")
ALGORITHM  = os.getenv("JWT_ALGORITHM", "HS256")
# int() con valor por defecto: si la variable no existe usa 30
ACCESS_TOKEN_EXPIRE_MINUTES = int(
    os.getenv("ACCESS_TOKEN_EXPIRE_MINUTES", "30")
)

def crear_token_acceso(data: dict) -> str:
    datos = data.copy()
    datos["exp"] = datetime.now(timezone.utc) + timedelta(
        minutes=ACCESS_TOKEN_EXPIRE_MINUTES  # ← ya viene del .env
    )
    return jwt.encode(datos, SECRET_KEY, algorithm=ALGORITHM)

import pytest
from httpx import AsyncClient, ASGITransport
from datetime import date, timedelta
from app.main import app

# Helper: obtener token de admin
async def get_token(ac: AsyncClient, email: str, password: str) -> str:
    r = await ac.post("/login",
        data={"username": email, "password": password})
    return r.json().get("access_token", "")


@pytest.mark.asyncio
async def test_cp_curso_01_inscripcion_exitosa():
    """CP-CURSO-01: un usuario se inscribe antes de la fecha de inicio"""
    async with AsyncClient(
        transport=ASGITransport(app=app), base_url="http://test"
    ) as ac:
        token = await get_token(ac, "sofia@sena.edu.co", "sena2026")
        headers = {"Authorization": f"Bearer {token}"}

        # Crear curso con fecha de inicio en el futuro
        fecha_inicio = str(date.today() + timedelta(days=30))
        curso_r = await ac.post("/cursos",
            json={"nombre": "Python Test", "capacidad": 5,
                  "fecha_inicio": fecha_inicio},
            headers=headers)
        assert curso_r.status_code == 201
        curso_id = curso_r.json()["id"]

        # Inscribir usuario 1 con fecha de hoy (anterior al inicio)
        insc_r = await ac.post(
            f"/cursos/{curso_id}/inscribir/1",
            json={"fecha_inscripcion": str(date.today())},
            headers=headers)

        assert insc_r.status_code == 200

Variante síncrona — cuándo usarla y cómo es diferente

Si tu proyecto tiene pocos usuarios simultáneos (un sistema interno, una API de clase, un prototipo) el modo síncrono es más simple: menos imports, sin await, sin asyncpg. Esta estación muestra cómo se ve el mismo backend en modo síncrono para que entiendas exactamente qué cambia y por qué en producción preferimos async.
